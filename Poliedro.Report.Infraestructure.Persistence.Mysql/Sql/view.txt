

CREATE OR REPLACE ALGORITHM = UNDEFINED VIEW `v_draft` AS
select
    cast(`t`.`created_at` as date) AS `fecha`,
    `p`.`sku` AS `codigo`,
    `p`.`name` AS `producto`,
    `ts`.`quantity` AS `cantidad`,
    `ts`.`unit_price_inc_tax` AS `precio`,
    (`ts`.`quantity` * `ts`.`unit_price_inc_tax`) AS `subtotal`,
    `c`.`name` AS `cliente`,
    `t`.`invoice_no` AS `referencia`
from
    (((`transactions` `t`
join `transaction_sell_lines` `ts` on
    ((`t`.`id` = `ts`.`transaction_id`)))
join `contacts` `c` on
    ((`t`.`contact_id` = `c`.`id`)))
join `products` `p` on
    ((`p`.`id` = `ts`.`product_id`)))
where
    (`t`.`status` = 'draft')
order by
    cast(`t`.`created_at` as date) desc;




CREATE OR REPLACE
ALGORITHM = UNDEFINED VIEW `v_inventario` AS
select
    `p`.`sku` AS `SKU`,
    `p`.`name` AS `Name`,
    `u`.`actual_name` AS `Presentation`,
    coalesce(max(`pc`.`purchase_price_inc_tax`), 0) AS `Cost`,
    coalesce(max(`v`.`sell_price_inc_tax`), coalesce(max(`pc`.`purchase_price_inc_tax`), 0)) AS `Sale`,
    max(`vd`.`qty_available`) AS `Inventory`,
    max(`v`.`profit_percent`) AS `Percentage`,
    if((coalesce(max(`pc`.`purchase_price_inc_tax`), 0) = 0), 0,(max(`vd`.`qty_available`) * coalesce(max(`pc`.`purchase_price_inc_tax`), 0))) AS `Subtotal_Cost`,
    if((coalesce(max(`pc`.`purchase_price_inc_tax`), 0) = 0), 0,(max(`vd`.`qty_available`) * coalesce(max(`v`.`sell_price_inc_tax`), coalesce(max(`pc`.`purchase_price_inc_tax`), 0)))) AS `Subtotal_Sale`
from
    ((((((`variation_location_details` `vd`
left join `products` `p` on
    ((`p`.`id` = `vd`.`product_id`)))
left join `categories` `c` on
    ((`c`.`id` = `p`.`category_id`)))
left join `units` `u` on
    ((`u`.`id` = `p`.`unit_id`)))
left join `purchase_lines` `pc` on
    ((`p`.`id` = `pc`.`product_id`)))
left join `transaction_sell_lines` `pv` on
    ((`pv`.`product_id` = `p`.`id`)))
left join `variations` `v` on
    ((`v`.`product_id` = `p`.`id`)))
group by
    `p`.`sku`,
    `p`.`name`,
    `u`.`actual_name`
order by
    `p`.`sku` desc;

ALGORITHM = UNDEFINED VIEW `v_invoice` AS
select
    `t`.`id` AS `id`,
    coalesce(`c`.`contact_id`, 'N/A') AS `identication`,
    coalesce(`c`.`name`, 'Consumidor Final') AS `contact_name`,
    coalesce(`c`.`email`, 'poliedrosoftware@gmail.com') AS `email`,
    coalesce(`c`.`mobile`, '3154286798') AS `mobile`,
    coalesce(`c`.`city`, 'Ocaña') AS `city`,
    coalesce(`c`.`state`, 'Norte de Santander') AS `state`,
    coalesce(`c`.`country`, 'Colombia') AS `country`,
    `t`.`invoice_no` AS `invoice`,
    (case
        when (`t`.`payment_status` in ('due', 'partial')) then 'credit'
        else 'counted'
    end) AS `payment_status`,
    `t`.`transaction_date` AS `transaction_date`,
    0 AS `allowanceTotal`,
    coalesce((select sum(`tsl`.`unit_price`) from `transaction_sell_lines` `tsl` where (`tsl`.`transaction_id` = `t`.`id`)), 0.0) AS `invoiceBaseTotal`,
    coalesce(`t`.`final_total`, 0.0) AS `invoiceTaxExclusiveTotal`,
    coalesce(`t`.`final_total`, 0.0) AS `invoiceTaxInclusiveTotal`,
    coalesce(`t`.`final_total`, 0.0) AS `totalToPay`,
    coalesce((select sum(`tsl`.`item_tax`) from `transaction_sell_lines` `tsl` where (`tsl`.`transaction_id` = `t`.`id`)), 0.0) AS `invoiceTax`,
    coalesce(`tax_info`.`total_tax_amount`, 0.0) AS `percent`,
    coalesce(`u`.`username`, 'Vendedor') AS `created_by_name`
from
    (((`transactions` `t`
left join `contacts` `c` on
    ((`t`.`contact_id` = `c`.`id`)))
left join `users` `u` on
    ((`t`.`created_by` = `u`.`id`)))
left join (
    select
        `tsl`.`transaction_id` AS `transaction_id`,
        sum(`tr`.`amount`) AS `total_tax_amount`
    from
        (`transaction_sell_lines` `tsl`
    left join `tax_rates` `tr` on
        ((`tsl`.`tax_id` = `tr`.`id`)))
    group by
        `tsl`.`transaction_id`) `tax_info` on
    ((`tax_info`.`transaction_id` = `t`.`id`)))
where
    (`t`.`type` = 'sell')
order by
    `t`.`created_at` desc;

from

left join `products` `p` on
    ((`tsl`.`product_id` = `p`.`id`)))
left join `variations` `v` on
    ((`tsl`.`variation_id` = `v`.`id`)))
order by
    `tsl`.`transaction_id` desc;




CREATE OR REPLACE
ALGORITHM = UNDEFINED VIEW `v_sales_for_pay_method` AS
select
    `v`.`transaction_id` AS `transaction_id`,
    cast(min(`v`.`created_at`) as date) AS `date`,
    monthname(min(`v`.`created_at`)) AS `month`,
    month(min(`v`.`created_at`)) AS `nmonth`,
    year(min(`v`.`created_at`)) AS `year`,
    `t`.`invoice_no` AS `bill`,
    sum(distinct (`v`.`quantity` * `v`.`unit_price_inc_tax`)) AS `subtotal`,
    (case
        when (`tp`.`method` = 'other') then 'otros'
        when (`tp`.`method` = 'cash') then 'efectivo'
        when (`tp`.`method` = 'card') then 'tarjeta'
        when (`tp`.`method` = 'bank_transfer') then 'transferencia'
        else `tp`.`method`
    end) AS `payment_method`,
    (case
        when (`tp`.`is_return` = 1) then -(`tp`.`amount`)
        else `tp`.`amount`
    end) AS `payment_amount`
from
    ((`transaction_sell_lines` `v`
left join `transactions` `t` on
    ((`t`.`id` = `v`.`transaction_id`)))
left join `transaction_payments` `tp` on
    ((`tp`.`transaction_id` = `v`.`transaction_id`)))
where
    (`tp`.`method` is not null)
group by
    `v`.`transaction_id`,
    `tp`.`id`,
    `tp`.`method`,
    `tp`.`amount`,
    `t`.`invoice_no`
order by
    `v`.`transaction_id`;

    `v`.`payment_method`

CREATE OR REPLACE
ALGORITHM = UNDEFINED VIEW `v_utility_for_sales_for_product` AS
select
    cast(`v`.`created_at` as date) AS `date`,
    `v`.`product_id` AS `productid`,
    `p`.`name` AS `product`,
    `vc`.`quantity` AS `amount`,
    `c`.`purchase_price_inc_tax` AS `cost`,
    `v`.`unit_price_inc_tax` AS `sales_price`,
    ((`v`.`unit_price_inc_tax` - `c`.`purchase_price_inc_tax`) * `vc`.`quantity`) AS `utility`
from
    (((`transaction_sell_lines_purchase_lines` `vc`
join `transaction_sell_lines` `v` on
    ((`v`.`id` = `vc`.`sell_line_id`)))
join `purchase_lines` `c` on
    ((`c`.`id` = `vc`.`purchase_line_id`)))
join `products` `p` on
    ((`p`.`id` = `v`.`product_id`)))
order by
    cast(`v`.`created_at` as date) desc;

select
    `subquery`.`year` AS `year`,
    `subquery`.`month` AS `month`,
    sum(`subquery`.`utility`) AS `utility`
from
    (
    select
        `v_utility_for_day`.`year` AS `year`,
        `v_utility_for_day`.`month` AS `month`,
        `v_utility_for_day`.`utility` AS `utility`,
        date_format(`v_utility_for_day`.`date`, '%Y-%m') AS `month_date`
    from
        `v_utility_for_day`) `subquery`
group by
    `subquery`.`year`,
    `subquery`.`month`,
    `subquery`.`month_date`
order by
    `subquery`.`year` desc,
    `subquery`.`month_date` desc;
